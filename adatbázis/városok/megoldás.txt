select vnev,nepesseg from varos
inner join megye on varos.megyeid=megye.id

select varostipus.vtip as "Város Típús", (select count(varos.id)) as "Városok Száma", sum(varos.nepesseg) as "Népesség" 
from varos
inner join varostipus on varostipus.id=varos.vtipid
where varostipus.vtip != "főváros"
group by varostipus.vtip

select mnev, count(varos.id) as "db" 
from varos, megye
where megye.id=megyeid
and jaras not like kisterseg
group by mnev
having db>8
order by db desc
