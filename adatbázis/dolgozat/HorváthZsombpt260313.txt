select (select sum(nepesseg) from orszagok where foldr_hely like "%Európa%") as "Európa népessége",
(select sum(nepesseg) from orszagok where foldr_hely like "%Afrika%") as "Afrika népessége",
(select sum(nepesseg) from orszagok where foldr_hely like "%Ázsia%") as "Ázsia népessége",
(select sum(nepesseg) from orszagok where foldr_hely like "%Észak-Amerika%" or "%Közép-Amerika%") as "Észak-Amerika népessége",
(select sum(nepesseg) from orszagok where foldr_hely like "%Dél-Amerika%") as "Dél-Amerika népessége",
(select sum(nepesseg) from orszagok where foldr_hely like "%Óceánia%" or "%Ausztrália%") as "Óceánia és Ausztrália népessége"
